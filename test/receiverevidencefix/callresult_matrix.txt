# arm (L) fix round 3 — THE BINDING-FORM MATRIX (read by test/receiverevidencecheck.sh; Python, kept as .txt so the repo's own
# indexed map does not grow — the same reason callresult.txt is a text bundle).
#
#   python3 -I callresult_matrix.txt OUTDIR     writes one tree per language under OUTDIR/mx_<lang>/ and prints the cells:
#     CELL <id> <lang> <form> <scope> <target> <root> <selector> P <kind> <name> <file:line>   (the ONE proven row)
#     CELL <id> <lang> <form> <scope> <target> <root> <selector> N <name> [FLOOR]              (no row <name> proven)
#     NA   <lang> <form> <scope> <target> <reason…>                                            (the language cannot say it)
#
# THE CLASS. A written (or returned) type name is resolved to an in-tree class while some binding visible at the callee's
# scope names that spelling as something else. Each cell writes ONE binding form of the spelling S<k> at ONE scope level of
# the callee load<k>(), with ONE target, and plants a TRAP — an in-tree namesake S<k> (with the member hit<k>) where a
# scope-blind or form-blind lookup would find it — plus a decoy D<k>.hit<k> so a by-name pick is never single. The caller
# writes `x = load<k>(); x.hit<k>()`.
#   target U  every visible binding names the same in-tree class → the cell's tagged hit<k> is the ONE proven row
#   target O  the deciding binding names something OUTSIDE the tree → no hit<k> is proven (the trap never wins)
#   target X  the deciding binding is UNRESOLVED (an outside wildcard, a path naming no in-tree module/class) → none proven
#   target 2  two visible bindings (or two in-tree classes) disagree → none proven
#   FLOOR     a U cell the rule does not follow (an alias, a member type inherited from a base, …): it must HEDGE, never
#             prove — the gate header names each floor.
# Scope levels: file, namespace / module (Rust inline `mod`, C++ / TS `namespace`, a JVM package-level class), nested
# (a mod in a mod, a C++ class or nested namespace, a JVM nested class / object / companion), function body.
# A binding is visible where the LANGUAGE puts it in force: an inner scope's binding shadows an outer one (a Rust `mod` is a
# fresh scope; a JVM nested class hides the package's), and a lower tier is read only when every higher one is silent
# (JVM: nested, single-type import / own file, package, on-demand; Rust: item / `use` / alias, then globs). Within the
# deciding tier every binding must agree and stay in tree.
import os, sys

OUT = sys.argv[ 1 ]
files = {}      # root -> { path: text }
# C++ OFF (owner 2026-10-17, fix round 6): model.h callResultFamilyOf maps C++ / ObjC to None, so no C++ call result is typed —
# every C++ U cell is printed as an N FLOOR (it must hedge). True again with the macro-aware C++ card.
CPP_TYPED = False
cells = []      # ( id, lang, form, scope, target, root, selector, verdict, kind, name, tag, floor )
nas   = []
K     = [ 0 ]

def new_id():
    K[ 0 ] += 1
    return K[ 0 ]

def put( root, path, text ):
    files.setdefault( root, {} )
    files[ root ][ path ] = files[ root ].get( path, "" ) + text

def cell( lang, form, scope, target, root, sel, k, expect, kind="method", floor=False ):
    cells.append( ( k, lang, form, scope, target, root, sel, expect, kind, "hit%d" % k if lang not in ( "go", ) else "Hit%d" % k, "@T%d" % k, floor ) )

def na( lang, form, scope, target, why ):
    nas.append( ( lang, form, scope, target, why ) )

def ind( text, n ):
    pad = " " * n
    return "".join( ( pad + l if l.strip() else l ) for l in text.splitlines( True ) )

# ───────────────────────────────────────────────────────────── Rust ─────────────────────────────────────────────────────────
RS = "mx_rust"
put( RS, "Cargo.toml", '[package]\nname = "mxr"\nversion = "0.1.0"\n' )
rs_mods = [ "decoy" ]

def rs_class( name, k, tag=False ):
    return "pub struct %s;\nimpl %s {\n    pub fn hit%d(&self) -> usize { 1 }%s\n}\n" % ( name, name, k, " // @T%d" % k if tag else "" )

def rs_mod( k, name, body ):
    rs_mods.append( name )
    put( RS, "src/%s.rs" % name, body )

def rs_cell( form, scope, target, k, bind, callee_ret, outer, expect, floor=False, extra_file="" ):
    """bind: the binding lines at the callee's level; outer: the trap at the enclosing level ('' none)."""
    S = "S%d" % k
    callee = "pub fn load%d() -> %s { todo!() }\n" % ( k, callee_ret or S )
    if scope == "file":
        body = bind + callee + "pub fn cell%d() -> usize { let x = load%d(); x.hit%d() }\n" % ( k, k, k )
        if outer:
            rs_mod( k, "w%d" % k, outer )    # the trap: a module nothing imports
    elif scope == "mod":
        body = outer + "pub mod m {\n" + ind( bind + callee, 4 ) + "}\n" + "pub fn cell%d() -> usize { let x = m::load%d(); x.hit%d() }\n" % ( k, k, k )
    elif scope == "nested":
        body = "pub mod m {\n" + ind( outer, 4 ) + "    pub mod n {\n" + ind( bind + callee, 8 ) + "    }\n}\n" \
             + "pub fn cell%d() -> usize { let x = m::n::load%d(); x.hit%d() }\n" % ( k, k, k )
    else:   # fn body: the callee is a fn nested in the caller's body; the module's items are in scope there
        body = outer + "pub fn cell%d() -> usize {\n" % k + ind( bind + callee.replace( "pub fn", "fn" ), 4 ) \
             + "    let x = load%d();\n    x.hit%d()\n}\n" % ( k, k )
    rs_mod( k, "c%d" % k, extra_file + body )
    cell( "rust", form, scope, target, RS, "src/c%d.rs:cell%d" % ( k, k ), k, expect, floor=floor )

for scope in ( "file", "mod", "nested", "fn" ):
    # item: a struct declared at the callee's own level; the trap one level out (file: another module)
    k = new_id(); rs_cell( "item", scope, "U", k, rs_class( "S%d" % k, k, True ), "", rs_class( "S%d" % k, k ), "P" )
    # none: nothing at the callee's level binds the name — the trap one level out is NOT in scope (a mod is a fresh
    # scope), except in a fn body, which sees its module's items
    k = new_id()
    if scope == "fn":
        rs_cell( "none", scope, "U", k, "", "", rs_class( "S%d" % k, k, True ), "P" )
    else:
        rs_cell( "none", scope, "X", k, "", "", rs_class( "S%d" % k, k ), "N" )
    # use: an explicit `use` at the callee's level
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k, True ) )
    rs_cell( "use", scope, "U", k, "use crate::t%d::S%d;\n" % ( k, k ), "", rs_class( "S%d" % k, k ), "P" )
    k = new_id(); rs_cell( "use", scope, "O", k, "use ext_crate::S%d;\n" % k, "", rs_class( "S%d" % k, k ), "N" )
    k = new_id(); rs_cell( "use", scope, "X", k, "use crate::nowhere%d::S%d;\n" % ( k, k ), "", rs_class( "S%d" % k, k ), "N" )
    na( "rust", "use", scope, "2", "two `use`s of one name in one scope (E0252) or a `use` beside an item of that name (E0255) do not compile" )
    # use … as
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "Real%d" % k, k, True ) )
    rs_cell( "use-as", scope, "U", k, "use crate::t%d::Real%d as S%d;\n" % ( k, k, k ), "", rs_class( "S%d" % k, k ), "P" )
    k = new_id(); rs_cell( "use-as", scope, "O", k, "use ext_crate::Thing as S%d;\n" % k, "", rs_class( "S%d" % k, k ), "N" )
    k = new_id(); rs_cell( "use-as", scope, "X", k, "use crate::nowhere%d::Real%d as S%d;\n" % ( k, k, k ), "", rs_class( "S%d" % k, k ), "N" )
    na( "rust", "use-as", scope, "2", "as for `use`: E0252 / E0255" )
    # glob
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k, True ) )
    rs_cell( "glob", scope, "U", k, "use crate::t%d::*;\n" % k, "", rs_class( "S%d" % k, k ), "P" )
    k = new_id(); rs_cell( "glob", scope, "O", k, "use ext_crate::*;\n", "", rs_class( "S%d" % k, k ), "N" )
    k = new_id(); rs_cell( "glob", scope, "X", k, "use crate::nowhere%d::*;\n" % k, "", rs_class( "S%d" % k, k ), "N" )
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k ) ); rs_mod( k, "u%d" % k, rs_class( "S%d" % k, k ) )
    rs_cell( "glob", scope, "2", k, "use crate::t%d::*;\nuse crate::u%d::*;\n" % ( k, k ), "", "", "N" )
    # alias: `type S = …;` is an item — it hides a glob's S; it is not followed (an in-tree target is a FLOOR)
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "Real%d" % k, k, True ) )
    rs_cell( "alias", scope, "U", k, "pub type S%d = crate::t%d::Real%d;\n" % ( k, k, k ), "", rs_class( "S%d" % k, k ), "N", floor=True )
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k ) )
    rs_cell( "alias", scope, "O", k, "pub type S%d = ext_crate::Thing;\nuse crate::t%d::*;\n" % ( k, k ), "", rs_class( "S%d" % k, k ), "N" )
    na( "rust", "alias", scope, "X", "an alias's target is the alias's own business: O covers an outside one, U (floor) an in-tree one" )
    na( "rust", "alias", scope, "2", "two items of one name in one scope do not compile (E0428)" )
    na( "rust", "item", scope, "O", "an item is a declaration in the tree" )
    na( "rust", "item", scope, "X", "an item is a declaration in the tree" )
    na( "rust", "item", scope, "2", "two items of one name in one scope do not compile (E0428)" )
for scope in ( "file", "mod" ):
    # an item shadows a glob of the same level
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k ) )
    rs_cell( "item+glob", scope, "U", k, rs_class( "S%d" % k, k, True ) + "use crate::t%d::*;\n" % k, "", "", "P" )
# a type parameter of the enclosing impl names the spelling: `impl<S> H<S> { fn load() -> S }` beside a struct S
for scope in ( "file", "mod" ):
    k = new_id(); S = "S%d" % k
    impl = "pub struct H%d<%s>(pub %s);\nimpl<%s> H%d<%s> {\n    pub fn load%d() -> %s { todo!() }\n}\n" % ( k, S, S, S, k, S, k, S )
    if scope == "file":
        body = rs_class( S, k ) + impl + "pub fn cell%d() -> usize { let x = H%d::<u8>::load%d(); x.hit%d() }\n" % ( k, k, k, k )
    else:
        body = "pub mod m {\n" + ind( rs_class( S, k ) + impl, 4 ) + "}\npub fn cell%d() -> usize { let x = m::H%d::<u8>::load%d(); x.hit%d() }\n" % ( k, k, k, k )
    rs_mod( k, "c%d" % k, body )
    cell( "rust", "impl-type-param", scope, "O", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "N" )
# a `use` in a scope that does not hold the callee is not in force there (the file's outside glob decides: nothing)
for where in ( "mod", "fn" ):
    k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k ) )
    side = ( "pub mod side {\n    use crate::t%d::S%d;\n}\n" if where == "mod" else "pub fn side() {\n    use crate::t%d::S%d;\n}\n" ) % ( k, k )
    rs_cell( "use-elsewhere(%s)" % where, "file", "X", k, "use ext_crate::*;\n" + side, "", "", "N" )
# qualified results written from inside an inline mod: `super::` / `self::` are relative to the mod, not the file
k = new_id(); rs_mods.append( "c%d" % k )   # super:: from the file module = the crate root
put( RS, "src/lib.rs", rs_class( "S%d" % k, k, True ) )
put( RS, "src/c%d.rs" % k, "pub fn load%d() -> super::S%d { todo!() }\npub fn cell%d() -> usize { let x = load%d(); x.hit%d() }\n" % ( k, k, k, k, k ) )
rs_mod( k, "w%d" % k, rs_class( "S%d" % k, k ) )
cell( "rust", "path super::", "file", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "P" )
k = new_id(); rs_mods.append( "c%d" % k )   # super:: from an inline mod = the file's own module
put( RS, "src/lib.rs", rs_class( "S%d" % k, k ) )
put( RS, "src/c%d.rs" % k, rs_class( "S%d" % k, k, True ) + "pub mod m {\n    pub fn load%d() -> super::S%d { todo!() }\n}\npub fn cell%d() -> usize { let x = m::load%d(); x.hit%d() }\n" % ( k, k, k, k, k ) )
cell( "rust", "path super::", "mod", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "P" )
k = new_id(); rs_mods.append( "c%d" % k )   # super:: from a mod in a mod = the outer inline mod
put( RS, "src/lib.rs", rs_class( "S%d" % k, k ) )
put( RS, "src/c%d.rs" % k, rs_class( "S%d" % k, k ) + "pub mod m {\n" + ind( rs_class( "S%d" % k, k, True ), 4 )
     + "    pub mod n {\n        pub fn load%d() -> super::S%d { todo!() }\n    }\n}\npub fn cell%d() -> usize { let x = m::n::load%d(); x.hit%d() }\n" % ( k, k, k, k, k ) )
cell( "rust", "path super::", "nested", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "P" )
k = new_id(); rs_mods.append( "c%d" % k )   # self:: inside an inline mod = that mod
put( RS, "src/c%d.rs" % k, rs_class( "S%d" % k, k ) + "pub mod m {\n" + ind( rs_class( "S%d" % k, k, True ), 4 )
     + "    pub fn load%d() -> self::S%d { todo!() }\n}\npub fn cell%d() -> usize { let x = m::load%d(); x.hit%d() }\n" % ( k, k, k, k, k ) )
cell( "rust", "path self::", "mod", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "P" )
k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k, True ) )   # crate:: is absolute anywhere
rs_mod( k, "c%d" % k, rs_class( "S%d" % k, k ) + "pub mod m {\n    pub fn load%d() -> crate::t%d::S%d { todo!() }\n}\npub fn cell%d() -> usize { let x = m::load%d(); x.hit%d() }\n" % ( k, k, k, k, k, k ) )
cell( "rust", "path crate::", "mod", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "P" )
k = new_id(); rs_mod( k, "t%d" % k, rs_class( "S%d" % k, k ) )   # a file-level `use crate::t;` is not in force inside `mod m`
rs_mod( k, "c%d" % k, "use crate::t%d;\npub mod m {\n    pub fn load%d() -> t%d::S%d { todo!() }\n}\npub fn cell%d() -> usize { let x = m::load%d(); x.hit%d() }\n" % ( k, k, k, k, k, k, k ) )
cell( "rust", "path (file use)", "mod", "X", RS, "src/c%d.rs:cell%d" % ( k, k ), k, "N" )
put( RS, "src/lib.rs", "".join( "pub mod %s;\n" % m for m in rs_mods ) )

# ───────────────────────────────────────────────────────────── C++ ──────────────────────────────────────────────────────────
CP = "mx_cpp"

def cp_class( name, k, tag=False ):
    return "struct %s {\n    int hit%d() const { return 1; }%s\n};\n" % ( name, k, " // @T%d" % k if tag else "" )

def cp_cell( form, scope, target, k, pre, bind, expect, trap_tu=False, floor=False, ret=None ):
    """pre: header text before the callee's scope (in-tree classes); bind: the binding at the callee's level."""
    S = ret or "S%d" % k
    if scope == "global":
        hdr = pre + bind + "%s load%d();\n" % ( S, k )
        src = "%s load%d() { return {}; }\n" % ( S, k )
        call = "load%d()" % k
    elif scope == "namespace":
        hdr = pre + "namespace n%d {\n" % k + bind + "%s load%d();\n}\n" % ( S, k )
        src = "namespace n%d {\n%s load%d() { return {}; }\n}\n" % ( k, S, k )
        call = "n%d::load%d()" % ( k, k )
    elif scope == "nested-ns":
        hdr = pre + "namespace n%d {\n" % k + bind + "namespace inner {\n%s load%d();\n}\n}\n" % ( S, k )
        src = "namespace n%d {\nnamespace inner {\n%s load%d() { return {}; }\n}\n}\n" % ( k, S, k )
        call = "n%d::inner::load%d()" % ( k, k )
    else:   # class: the callee is a static member defined in the class; the binding is a member
        hdr = pre + "struct H%d {\n" % k + ind( bind, 4 ) + "    static %s load%d() { return {}; }\n};\n" % ( S, k )
        src = ""
        call = "H%d::load%d()" % ( k, k )
    put( CP, "src/c%d.h" % k, "#pragma once\n#include <ext/things.h>\n" + hdr )
    put( CP, "src/c%d.cpp" % k, '#include "c%d.h"\n' % k + src + "int cell%d() { auto x = %s; return x.hit%d(); }\n" % ( k, call, k ) )
    if trap_tu:
        put( CP, "src/w%d.cpp" % k, cp_class( "S%d" % k, k ) )   # a TU nothing includes: the in-tree namesake
    put( CP, "src/decoy.cpp", cp_class( "D%d" % k, k ) )
    cell( "cpp", form, scope, target, CP, "src/c%d.cpp:cell%d" % ( k, k ), k, expect, floor=floor )

# none: the class is a member of a scope enclosing the callee (Rule 2 keys members by NAME: the class must be the only one)
k = new_id(); cp_cell( "none", "global", "U", k, cp_class( "S%d" % k, k, True ), "", "P" )
k = new_id(); cp_cell( "none", "namespace", "U", k, "namespace n%d {\n" % k + cp_class( "S%d" % k, k, True ) + "}\n", "", "P" )
k = new_id(); cp_cell( "none", "nested-ns", "U", k, "namespace n%d {\n" % k + cp_class( "S%d" % k, k, True ) + "}\n", "", "P" )
k = new_id(); cp_cell( "none", "class", "U", k, "", cp_class( "S%d" % k, k, True ), "P" )
k = new_id(); cp_cell( "none", "global", "X", k, "namespace o%d {\n" % k + cp_class( "S%d" % k, k ) + "}\n", "", "N" )
k = new_id(); cp_cell( "none", "namespace", "X", k, "namespace o%d {\n" % k + cp_class( "S%d" % k, k ) + "}\n", "", "N" )
k = new_id(); cp_cell( "none", "namespace", "2", k, "namespace n%d {\n" % k + cp_class( "S%d" % k, k ) + "}\nnamespace o%d {\n" % k + cp_class( "S%d" % k, k ) + "}\n", "", "N", floor=True )
na( "cpp", "none", "class", "X", "a class member is in tree; the class-scope traps are the alias/typedef/inherited rows" )
# using-declaration
for scope in ( "global", "namespace", "nested-ns" ):
    k = new_id(); cp_cell( "using-decl", scope, "U", k, "namespace lib%d {\n" % k + cp_class( "S%d" % k, k, True ) + "}\n", "using lib%d::S%d;\n" % ( k, k ), "P" )
    k = new_id(); cp_cell( "using-decl", scope, "O", k, "" if scope == "global" else cp_class( "S%d" % k, k ), "using ext::S%d;\n" % k, "N", trap_tu=( scope == "global" ) )
    k = new_id(); cp_cell( "using-decl", scope, "X", k, "namespace lib%d {\nint helper%d();\n}\n" % ( k, k ), "using lib%d::S%d;\n" % ( k, k ), "N", trap_tu=True )
    na( "cpp", "using-decl", scope, "2", "two using-declarations of one name naming different entities in one scope are ill-formed" )
na( "cpp", "using-decl", "class", "*", "a class-scope using-declaration names a BASE member (the inherited rows)" )
# using-directive
for scope in ( "global", "namespace", "nested-ns" ):
    k = new_id(); cp_cell( "using-directive", scope, "U", k, "namespace lib%d {\n" % k + cp_class( "S%d" % k, k, True ) + "}\n", "using namespace lib%d;\n" % k, "P" )
    k = new_id(); cp_cell( "using-directive", scope, "O", k, "namespace o%d {\n" % k + cp_class( "S%d" % k, k ) + "}\n", "using namespace ext;\n", "N" )
    k = new_id(); cp_cell( "using-directive", scope, "X", k, "namespace lib%d {\nint helper%d();\n}\nnamespace o%d {\n" % ( k, k, k ) + cp_class( "S%d" % k, k ) + "}\n", "using namespace lib%d;\n" % k, "N" )
k = new_id(); cp_cell( "using-directive", "namespace", "2", k, "namespace lib%d {\n" % k + cp_class( "S%d" % k, k ) + "}\nnamespace lib2_%d {\n" % k + cp_class( "S%d" % k, k ) + "}\n",
                       "using namespace lib%d;\nusing namespace lib2_%d;\n" % ( k, k ), "N" )
# an enclosing namespace's class beside an outside directive: lookup would be ambiguous if ext supplied the name (ill-formed)
k = new_id(); cp_cell( "using-directive+enclosing", "namespace", "U", k, "namespace n%d {\n" % k + cp_class( "S%d" % k, k, True ) + "}\n", "using namespace ext;\n", "P" )
na( "cpp", "using-directive", "class", "*", "a using-directive is not allowed at class scope" )
# alias-declaration / typedef (D5): an alias is a binding of the spelling; an outside target names nothing, an in-tree one is a FLOOR
for form, spell in ( ( "alias", "using S%d = %s;\n" ), ( "typedef", "typedef %s S%d;\n" ) ):
    def b( k, target ):
        return ( spell % ( k, target ) ) if form == "alias" else ( spell % ( target, k ) )
    for scope in ( "global", "namespace", "nested-ns", "class" ):
        k = new_id(); cp_cell( form, scope, "O", k, "" if scope == "global" else cp_class( "S%d" % k, k ), b( k, "ext::IntList" ), "N", trap_tu=( scope == "global" ) )
    k = new_id(); cp_cell( form, "namespace", "U", k, "namespace lib%d {\n" % k + cp_class( "Real%d" % k, k, True ) + "}\n", b( k, "lib%d::Real%d" % ( k, k ) ), "N", floor=True )
    k = new_id(); cp_cell( form, "global", "X", k, "namespace lib%d {\nint helper%d();\n}\n" % ( k, k ), b( k, "lib%d::Nope%d" % ( k, k ) ), "N", trap_tu=True )
    na( "cpp", form, "*", "2", "two aliases of one name in one scope are ill-formed unless they name the same type" )
# template parameters: of the enclosing class template, and of the callee itself
k = new_id(); S = "S%d" % k
put( CP, "src/c%d.h" % k, "#pragma once\n" + cp_class( S, k ) + "template <class %s> struct H%d {\n    static %s load%d() { return {}; }\n};\n" % ( S, k, S, k ) )
put( CP, "src/c%d.cpp" % k, '#include "c%d.h"\nint cell%d() { auto x = H%d<int>::load%d(); return x.hit%d(); }\n' % ( k, k, k, k, k ) )

def ts_cell( form, scope, target, k, top, bind, expect, floor=False, ret=None, callee=None, call=None ):
    S = ret or "S%d" % k
    if callee is None:
        if scope == "file":
            callee = bind + "export function load%d(): %s { return null as any; }\n" % ( k, S )
            imp, call = "load%d" % k, call or "load%d()" % k
        elif scope == "namespace":   # the caller sits in the namespace too (a bare call of its own function)
            callee = "export namespace n%d {\n" % k + ind( bind + "export function load%d(): %s { return null as any; }\n" % ( k, S )
                     + "export function cell%d(): number {\n    const x = load%d();\n    return x.hit%d();\n}\n" % ( k, k, k ), 4 ) + "}\n"
            imp, call = "n%d" % k, None
        else:   # fn body: a function nested in the exported loader's body
            callee = "export function run%d(): number {\n" % k + ind( bind + "function load%d(): %s { return null as any; }\nconst x = load%d();\nreturn x.hit%d();\n" % ( k, S, k, k ), 4 ) + "}\n"
            imp, call = "run%d" % k, None
    else:
        imp = callee[ 1 ]; callee = callee[ 0 ]
    put( TS, "src/c%d.ts" % k, top + callee )
    if call is None:   # the fn-body shape calls inside the loader's own body; the namespace shape inside the namespace

# elaborated class-key declaration of a bare name (`struct API S;`), a type-less function-style declaration that is not a
# constructor (`MK(S);` in a class body), and — for a local class's members — any statement of an enclosing function body
# that may be a macro invocation. Each may declare ANY name in its scope, so it hides every in-tree namesake there.
for scope in ( "namespace", "nested-ns", "class" ):
    k = new_id(); cp_cell( "unparsed:export-macro", scope, "X", k, cp_class( "S%d" % k, k ) + "#define APIM%d\n" % k,
                           "class APIM%d S%d { public: int hit%d() const { return 2; } };\n" % ( k, k, k ), "N" )
k = new_id(); cp_cell( "unparsed:export-macro(final)", "namespace", "X", k, cp_class( "S%d" % k, k ) + "#define APIM%d\n" % k,
                       "struct APIM%d S%d final { int hit%d() const { return 2; } };\n" % ( k, k, k ), "N" )
k = new_id(); cp_cell( "unparsed:export-macro(forward)", "namespace", "X", k, cp_class( "S%d" % k, k ) + "#define APIM%d\n" % k, "struct APIM%d S%d;\n" % ( k, k ), "N" )
k = new_id(); cp_cell( "unparsed:export-macro(leading)", "namespace", "X", k, cp_class( "S%d" % k, k ) + "#define APIM%d\n" % k,
                       "APIM%d struct S%d { int hit%d() const { return 2; } };\n" % ( k, k, k ), "N" )
k = new_id(); cp_cell( "unparsed:class-key-macro", "namespace", "X", k, cp_class( "S%d" % k, k ) + "#define MKS%d struct\n" % k,
                       "MKS%d S%d { int hit%d() const { return 2; } };\n" % ( k, k, k ), "N" )
for form, bind in ( ( "unparsed:class-body-macro", "MKC%d(S%d);\n" ), ( "unparsed:class-body-macro(2 args)", "MKC%d(S%d, int);\n" ) ):
    k = new_id(); cp_cell( form, "class", "X", k, cp_class( "S%d" % k, k ) + "#define MKC%d(...) struct S%d { int hit%d() const { return 2; } };\n" % ( k, k, k ), bind % ( k, k ), "N" )
# guard: constructors and a destructor are type-less declarations too — they declare no name
k = new_id(); cp_cell( "unparsed:none(constructors)", "class", "U", k, cp_class( "S%d" % k, k, True ), "H%d();\nexplicit H%d( int );\n~H%d();\n" % ( k, k, k ), "P" )
# C++ function body: a statement that may be a macro (any call statement) is in force for a local class's members
for form, block in ( ( "unparsed:fn-macro", "DECLF%d(S%d)\n" ), ( "unparsed:fn-macro;", "DECLF%d(S%d);\n" ), ( "unparsed:fn-macro(other name)", "REGF%d(Other%d);\n" ) ):
    k = new_id(); cp_local( form, "X", k, block % ( k, k ), "N" )
# guard: a function-body macro hides nothing outside that body (no local class: the namespace-scope callee is proven)
k = new_id(); cp_cell( "unparsed:fn-macro(no local class)", "namespace", "U", k, cp_class( "S%d" % k, k, True ) + "inline int helperF%d() { DECLF%d(S%d); return 0; }\n" % ( k, k, k ), "", "P" )
# guard (delta-4 M4): a forward declaration in the class's OWN namespace, reached through a using-declaration, declares that class
k = new_id(); S = "S%d" % k
put( CP, "src/c%d.h" % k, "#pragma once\nnamespace n%d {\nstruct %s;\nnamespace inner {\nusing n%d::%s;\n%s load%d();\n}\n}\nnamespace n%d {\n" % ( k, S, k, S, S, k, k )
     + cp_class( S, k, True ) + "}\n" )
put( CP, "src/c%d.cpp" % k, '#include "c%d.h"\nnamespace n%d {\nnamespace inner {\n%s load%d() { return {}; }\n}\n}\nint cell%d() { auto x = n%d::inner::load%d(); return x.hit%d(); }\n' % ( k, k, S, k, k, k, k, k ) )
put( CP, "src/decoy.cpp", cp_class( "D%d" % k, k ) )
cell( "cpp", "unrec:forward-decl+using-decl", "nested-ns", "U", CP, "src/c%d.cpp:cell%d" % ( k, k ), k, "P" )
# Rust — a method of an `impl` written in a fn body (directly, in an inner block, or of a module-level struct) sees that
# body's items, so a macro there may declare the spelling
def rs_impl_cell( form, k, body, expect, tag=False ):
    S = "S%d" % k
    rs_mod( k, "c%d" % k, rs_class( S, k, tag ) + ( "" if tag else rs_macro( k, "struct %s; impl %s { fn hit%d(&self) -> usize { 2 } }" % ( S, S, k ) ) ) + body % { "k": k, "S": S } )
    cell( "rust", form, "fn", "U" if tag else "X", RS, "src/c%d.rs:cell%d" % ( k, k ), k, expect )
IMPL = "    struct L%(k)d;\n    impl L%(k)d { fn load%(k)d() -> %(S)s { todo!() } }\n    let x = L%(k)d::load%(k)d();\n    x.hit%(k)d()\n"
k = new_id(); rs_impl_cell( "unparsed:impl-in-block", k, "pub fn cell%(k)d() -> usize {\n    mk%(k)d!();\n" + IMPL + "}\n", "N" )
k = new_id(); rs_impl_cell( "unparsed:impl-in-block(item macro)", k, "pub fn cell%(k)d() -> usize {\n    mk%(k)d! {}\n" + IMPL + "}\n", "N" )
k = new_id(); rs_impl_cell( "unparsed:impl-in-inner-block", k, "pub fn cell%(k)d() -> usize {\n    mk%(k)d!();\n    {\n" + ind( IMPL, 4 ) + "    }\n}\n", "N" )
k = new_id(); rs_impl_cell( "unparsed:impl-of-module-struct", k, "pub struct L%(k)d;\npub fn cell%(k)d() -> usize {\n    mk%(k)d!();\n"
                            "    impl L%(k)d { fn load%(k)d() -> %(S)s { todo!() } }\n    let x = L%(k)d::load%(k)d();\n    x.hit%(k)d()\n}\n", "N" )
k = new_id(); rs_impl_cell( "unparsed:none(impl-in-block)", k, "pub fn cell%(k)d() -> usize {\n" + IMPL + "}\n", "P", tag=True )
# fix round 6 (rv-call-result-typing-delta5 B3): a `const` / `static` item's value block is an item scope as well — the methods
# of an impl written in it see its items (a macro there, its own `use`); a const block with neither is no barrier (its own
# struct beside a module namesake splits: Rule 2 keys members by name — hedged, so no cell)
CIMPL = "    struct L%(k)d;\n    impl L%(k)d {\n        fn load%(k)d() -> %(S)s { todo!() }\n        #[allow(dead_code)]\n        fn cell%(k)d() -> usize { let x = Self::load%(k)d(); x.hit%(k)d() }\n    }\n"
def rs_const_cell( form, k, body, expect, tag=False, macro=True ):
    S = "S%d" % k
    rs_mod( k, "c%d" % k, rs_class( S, k, tag ) + ( rs_macro( k, "struct %s; impl %s { fn hit%d(&self) -> usize { 2 } }" % ( S, S, k ) ) if macro else "" ) + body % { "k": k, "S": S } )
    cells.append( ( k, "rust", form, "const-block", "U" if tag or expect == "P" else "X", RS, "src/c%d.rs:cell%d" % ( k, k ), expect, "method", "hit%d" % k, "@T%d" % k, False ) )
k = new_id(); rs_const_cell( "unparsed:const-block", k, "const _: () = {\n    mk%(k)d!();\n" + CIMPL + "};\n", "N" )
k = new_id(); rs_const_cell( "unparsed:static-block", k, "static X%(k)d: () = {\n    mk%(k)d!();\n" + CIMPL + "};\n", "N" )
k = new_id(); rs_const_cell( "unparsed:const-block-in-fn", k, "pub fn outer%(k)d() {\n    const _: () = {\n        mk%(k)d!();\n" + ind( CIMPL, 4 ) + "    };\n}\n", "N" )
k = new_id(); rs_const_cell( "unparsed:const-block(item macro)", k, "const C%(k)d: () = {\n    mk%(k)d! {}\n" + CIMPL + "};\n", "N" )
k = new_id(); rs_const_cell( "use:const-block-outside", k, "const _: () = {\n    use std::fmt::Error as %(S)s;\n" + CIMPL + "};\n", "N", macro=False )
k = new_id(); rs_const_cell( "unparsed:none(const-block)", k, "const _: () = {\n" + CIMPL + "};\n", "P", tag=True, macro=False )
# train 30 (N1, rv-call-result-typing-delta6 mutant M5): the qualified `use` walk (receiverevidence.h forEachRustUseAt) reads
# the same item levels — `use a as m;` at module level, `use b as m;` inside a const / static item's value block, and the
# block's impl returns `m::S`: the block's own `use` decides, so b's S is the ONE proven row and a's S (the trap) never is
for form, opener in ( ( "use:const-block-qualified", "const _: () = {\n" ), ( "use:static-block-qualified", "static Q%(k)d: () = {\n" ) ):
    k = new_id(); S = "S%d" % k
    QIMPL = CIMPL.replace( "-> %(S)s {", "-> m::%(S)s {" )
    rs_mod( k, "c%d" % k, "pub mod a {\n" + ind( rs_class( S, k ), 4 ) + "}\npub mod b {\n" + ind( rs_class( S, k, True ), 4 ) + "}\n"
            + "use a as m;\n" + ( opener + "    use b as m;\n" + QIMPL + "};\n" ) % { "k": k, "S": S }
            + "pub fn keep%d() -> usize { m::%s.hit%d() }\n" % ( k, S, k ) )
    cells.append( ( k, "rust", form, "const-block", "U", RS, "src/c%d.rs:cell%d" % ( k, k ), "P", "method", "hit%d" % k, "@T%d" % k, False ) )
put( RS, "src/lib.rs", "".join( "pub mod %s;\n" % m for m in rs_mods if ( "pub mod %s;\n" % m ) not in files[ RS ][ "src/lib.rs" ] ) )

def where( root, tag ):
    for path, text in files[ root ].items():
        for i, line in enumerate( text.splitlines(), 1 ):
            if line.rstrip().endswith( tag ):
                return "%s:%d" % ( path, i )
    return None

for ( k, lang, form, scope, target, root, sel, expect, kind, name, tag, floor ) in cells:
    f = lambda s: s.replace( " ", "_" )
    head = "CELL %d %s %s %s %s %s %s" % ( k, lang, f( form ), f( scope ), target, root, sel )
    if expect == "P" and lang == "cpp" and not CPP_TYPED:
        expect, floor = "N", True   # C++ off: the cell must hedge (its tagged row is not proven)
    if expect == "P":
        w = where( root, tag )
        if w is None:
            sys.exit( "cell %d: no %s tag" % ( k, tag ) )
        print( "%s P %s %s %s" % ( head, kind, name, w ) )
    else:
        print( "%s N %s%s" % ( head, name, " FLOOR" if floor else "" ) )
