// Book and Other both define spin / iter; Cfg and Knob both define tune: a call proves one only through its receiver's type.
pub struct Book { pub items: Vec<u32> }
impl Book {
    pub fn new() -> Self { Book { items: Vec::new() } }
    pub fn spin(&self) -> u8 { 1 }
    pub fn iter(&self) -> usize { 2 }
}
pub struct Other;
impl Other {
    pub fn spin(&self) -> u8 { 3 }
    pub fn iter(&self) -> usize { 4 }
}
#[derive(Default)]
pub struct Cfg;
impl Cfg {
    pub fn tune(&self) -> u8 { 5 }
}
pub struct Knob;
impl Knob {
    pub fn tune(&self) -> u8 { 6 }
}
pub struct Stack<T> { pub items: Vec<T> }
impl<T> Stack<T> {
    pub fn spin(&self) -> u8 { 7 }
}
pub struct Inner { pub book: Book }
pub struct Shelf {
    pub book: Book,
    pub boxed: Box<Book>,
    pub maybe: Option<Book>,
    pub shared: std::rc::Rc<Book>,
    pub inner: Inner,
    pub count: u32,
}
impl Shelf {
    pub fn iter(&self) -> usize { self.book.iter() }
    pub fn self_field(&self) -> u8 { self.book.spin() }
    pub fn self_boxed(&self) -> u8 { self.boxed.spin() }
    pub fn self_two_hops(&self) -> u8 { self.inner.book.spin() }
    pub fn self_literal(&self) -> u8 { let s = Self { book: Book::new(), boxed: Box::new(Book::new()), maybe: None, shared: todo!(), inner: todo!(), count: 0 }; s.book.spin() }
    pub fn self_param(&self, b: &Book) -> u8 { b.spin() }
    pub fn near_self_maybe(&self) -> u8 { self.maybe.spin() }
    pub fn near_self_shared(&self) -> u8 { self.shared.spin() }
    pub fn near_self_count(&self) -> u8 { self.count.spin() }
    pub fn near_self_unknown(&self) -> u8 { self.missing.spin() }
}
pub struct Holder<T> { pub item: T }
impl<T> Holder<T> {
    pub fn near_generic_field(&self) -> u8 { self.item.spin() }
}
pub struct Pair(pub Book);
impl Pair {
    pub fn near_tuple_field(&self) -> u8 { self.0.spin() }
}
